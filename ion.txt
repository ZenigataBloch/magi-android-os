name: Build APK

on:
  workflow_dispatch:
  push:
    branches: [main]
    paths:
      - "main.py"
      - "buildozer.spec"
      - ".github/workflows/build-apk.yml"

permissions:
  contents: write

concurrency:
  group: build-apk
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-22.04
    timeout-minutes: 90
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - uses: actions/setup-java@v5
        with:
          distribution: temurin
          java-version: "17"

      - name: Dipendenze di sistema
        run: |
          sudo apt-get update
          sudo apt-get install -y git zip unzip autoconf libtool pkg-config \
            zlib1g-dev libncurses-dev cmake libffi-dev libssl-dev

      - name: Cache SDK, NDK e build
        uses: actions/cache@v4
        with:
          path: |
            ~/.buildozer
            .buildozer
          key: buildozer-${{ hashFiles('buildozer.spec') }}
          restore-keys: buildozer-

      - name: Chiave di firma (stessa del Mint, cosi' l'app si aggiorna senza disinstallare)
        env:
          KS: ${{ secrets.DEBUG_KEYSTORE_B64 }}
        run: |
          mkdir -p ~/.android
          if [ -z "$KS" ]; then
            echo "DEBUG_KEYSTORE_B64 assente: firma con una chiave nuova"
            exit 0
          fi
          if ! printf '%s' "$KS" | tr -d '[:space:]' | base64 -d > ~/.android/debug.keystore; then
            echo "::error::DEBUG_KEYSTORE_B64 non e' base64 valido: rigenera il secret"
            exit 1
          fi
          keytool -list -keystore ~/.android/debug.keystore -storepass android > /dev/null \
            || { echo "::error::keystore non valido o con password diversa da 'android'"; exit 1; }

      - name: Installa Buildozer
        run: pip install buildozer cython

      - name: Compila
        run: buildozer -v android debug

      - name: Pubblica l'APK
        uses: softprops/action-gh-release@v2
        with:
          tag_name: apk-latest
          name: APK più recente
          body: "APK di debug compilato da GitHub Actions, senza chiavi API. Vedi il README per inserirle."
          files: bin/*.apk
